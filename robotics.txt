#             Plate 1                        Plate 2                          Plate 3
#   ┌────────┬────────┬────────┐   ┌────────┬────────┬────────┐   ┌─────────┬─────────┬─────────┐      ↑ Y
#   │[0,20]  │[10,20] │[20,20] │   │[60,20] │[70,20] │[80,20] │   │[100,20] │[110,20] │[120,20] │      |
#   ├────────┼────────┼────────┤   ├────────┼────────┼────────┤   ├─────────┼─────────┼─────────┤      |--→ X
#   │[0,10]  │[10,10] │[20,10] │   │[60,10] │[70,10] │[80,10] │   │[100,10] │[110,10] │[120,10] │      
#   ├────────┼────────┼────────┤   ├────────┼────────┼────────┤   ├─────────┼─────────┼─────────┤      [x,y]
#   │[0,0]   │[10,0]  │[20,0]  │   │[60,0]  │[70,0]  │[80,0]  │   │[100,0]  │[110,0]  │[120,0]  │
#   └────────┴────────┴────────┘   └────────┴────────┴────────┘   └─────────┴─────────┴─────────┘
#   ┌───┬───┬───┐   ┌────┬────┬────┐
#   │ l │ s │ m │   │ 0f │ mf │ ff │
#   ├───┼───┼───┤   ├────┼────┼────┤
#   │ m │ l │ s │   │ 0m │ mm │ mf │
#   ├───┼───┼───┤   ├────┼────┼────┤
#   │ s │ m │ l │   │ 0  │ m0 │ f0 │
#   └───┴───┴───┘   └────┴────┴────┘
#
# (SOURCE PLATE | SRC_DIST_X | SRC_DIST_Y | TARGET PLATE | TRG_DIST_X | TRG_DIST_Y)
# mv(src="A", src_x=0, src_y=f, trg="B", trg_x=m, trg_y=m) should move the bottom-left corner of the first plate to the second plate's middle [0,0, -> 70,10]
# 
# first itiration   (plate 1 to plate 2)
# [0,0,    ->  60,10]
# [10,10,  ->  70,0 ]
# [20,20,  ->  80,10]
# second itiration  (plate 2 to plate 3)
# [60,10,  ->  100,20]
# [70,0,   ->  100,10]
# [80,10,  ->  100,0 ]
# third itiration   (plate 3 to plate 2)
# [100,20, ->  80,20]
# [100,10, ->  70,10]
# [100,0,  ->  60,0 ]
# last itiration    (plate 2 to plate 1)
# [80,20,  ->  0,0  ]
# [70,10,  ->  10,10]
# [60,0,   ->  20,20]

c,o,a,v,tJ,tL = True, False, 100, 100, 1, 1
zu,zd = 265, 163
pCrn = { "A":0, "B":60, "C":100, }
m = 10
f = m*2

def grb(x):
    set_tool_digital_outputs([1, -2] if x else [-1, 2])
    wait(0.2)

def mSeq(x):
    for p1, p1a, p2, p2a in x:
        for p in [p1, p1a]: movel(p, v, a, tL)
        grb(c)
        movel(p1, v, a, tL)
        for p in [p2, p2a]: movel(p, v, a, tL)
        grb(o)
        movel(p2, v, a, tL)
        
def mv(src, src_x, src_y, trg, trg_x, trg_y):
    mSeq([[[pCrn[src]+src_x, src_y, z, 0,0,0] for z in [zu, zd]] + [[pCrn[trg]+trg_x, trg_y, z, 0,0,0] for z in [zu, zd]]])

movej([90,0,90,0,90,0], v,a,tJ) # align default position
grb(o)
# A → B
mv("A", 0, f, "B", m, f)
mv("A", m, f, "B", m, m)
mv("A", f, f, "B", m, 0) 
# B → C
mv("B", 0, m, "C", 0, f)
mv("B", m, m, "C", 0, m)
mv("B", f, m, "C", 0, 0)
# C → B
mv("C", 0, 0, "B", m, f) 
mv("C", m, 0, "B", m, m)
mv("C", f, 0, "B", m, 0)
# B → A
mv("B", 0, 0, "A", 0, f)
mv("B", m, 0, "A", 0, m) 
mv("B", f, 0, "A", 0, 0)
